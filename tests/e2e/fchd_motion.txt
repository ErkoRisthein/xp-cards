# FreeCell HD 2d (motion) end to end: the eased flights caught in mid-air, an autoplay cascade with two
# cards in the air at once (the next card takes off at 60% of the previous one's flight), the Finish
# button on the table once the rest is a sure win (game #31364, fchd_v12.txt's line) and its click;
# "Enhanced animations": the option, the dragged cards' shadow, the deal flying in. The mid-air captures
# run with FCHD_ANIM_SLOW=30 (every animation 30 times slower) and the driver's `async on` (the click
# returns at once; the capture is answered while the cards fly).  Run by tests/e2e/run_fchd_e2e.sh.
# The client is 1264 x 854, exactly scale 2 (columns x = 14 170 326 482 639 795 951 1107, first card
# y = 212, step 37, cards 142 x 192; free cells x = 0 142 284 426, home cells x = 696 838 980 1122,
# y 0..191; the Finish button 576,132 112 x 40, in the gap below the king).

timeout 60000
regdel HKCU\Software\Microsoft\Windows\CurrentVersion\Applets\FreeCell
regset HKCU\Software\Microsoft\Windows\CurrentVersion\Applets\FreeCell AlreadyPlayed bin32 1
regdel HKCU\Software\xp-cards
setenv FCHD_ANIM_SLOW 30
launch ${REPO}/build/FreeCellHD.exe
move_window 40 140
resize_client 1264 854
assert_client_size 1264 854
command 103
wait_dialog "Game Number"
dialog_set_text 203 1
dialog_click OK
assert_title "FreeCell Game #1"

# ---- a flight in mid-air: 3D (column 6) -> free cell 0, 120 ms (3.6 s at 30 x) -------------------------
click 866 300                             # select 3D (column 6, y 397..589)
async on
click 71 96                               # -> free cell 0: returns at once
wait 1000
capture 80_mid_flight.bmp
assert_pixel 71 96 007F00                 # not landed yet: free cell 0 still empty
assert_pixel 866 580 007F00               # lifted: 3D's place in column 6 is bare table
async off
sync
capture 81_landed.bmp
assert_pixel 71 96 FF0000                 # landed: 3D's middle pip in free cell 0

# ---- a cascade: 2C -> free cell 1, then autoplay AC, 2C, AS home, overlapping -----------------------
click 866 300                             # select 2C
async on
click 213 96                              # -> free cell 1; autoplay follows
wait 2200
capture 82_cascade_mid_air.bmp            # 2C nearly in free cell 1 while AC already flies home
assert_pixel 866 500 007F00               # AC has left column 6 (under it: AS, y 286..478)
async off
sync
capture 83_cascade_landed.bmp
assert_menubar_text "Cards Left: 49"
close Yes
