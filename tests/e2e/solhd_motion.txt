# Solitaire HD 2d (motion) end to end: the cards that go home automatically fly eased and overlap (two in
# the air at once), caught mid-air; "Enhanced animations": the option (off by default), the deal flying
# out of the stock, a card turning over (the flip), the dragged cards' shadow and the eased slide back of
# a refused drop. The mid-air captures run with SOLHD_ANIM_SLOW=30 (every animation 30 times slower) and
# the driver's `async on` (the input returns at once; the capture is answered while the cards fly).
# The Finish button is checked in solhd_extras.txt (deal 64's winning line); here, its double-click: the
# second press (it arrives while the cards fly home) never stops the win cascade. And a card turned over
# automatically that goes home at once (deal 1) first turns, then flies; the card under it stays.
# Run by run_solhd_e2e.sh.
#
# Deal 64 at 1264 x 854 (geometry in solhd_smoke.txt; cards 153 x 207, columns x = 24 201 378 555 732
# 909 1086, tableau top y = 231, steps 6 / 32; stock 24,10, waste 201,10, foundations x = 555 732 909
# 1086, y = 10): col 1 AH, col 2 #7H 4S, col 3 # # TD, col 4 # # # KC, col 5 # # # # 5D,
# col 6 # # # # # AC, col 7 # # # # # # 2H. Options 0x09: Draw Three, Standard, untimed. Extras 327
# AutoHome, 321 AutoTurn, 342 EnhancedAnimations.

timeout 60000
regdel HKCU\Software\Microsoft\Solitaire
regset HKCU\Software\Microsoft\Solitaire Back dword 1
regset HKCU\Software\Microsoft\Solitaire Options dword 9
regdel "HKCU\Software\xp-cards\Solitaire HD"
regset "HKCU\Software\xp-cards\Solitaire HD" AutoHome dword 1
setenv SOLHD_TIME 64
setenv SOLHD_ANIM_SLOW 30
launch ${REPO}/build/SolitaireHD.exe
sync                                          # (in its message loop before the first command)
move_window 0 120
resize_client 1264 854
assert_client_size 1264 854
command 1003
wait_dialog Options
assert_dialog_text "[342] Button \"Enhanced animations\" [ ]"   # off by default
capture_dialog 80_options_enhanced.bmp
dialog_click OK

# ---- a draw, then AH and AC fly home by themselves, overlapping ------------------------------------------
async on
click 100 100                                 # the stock: a draw; AH (col 1), AC (col 6), 2H go home
wait 900
capture 81_auto_home_mid_air.bmp
assert_pixel 100 400 008000                   # AH has left col 1
async off
sync
capture 82_auto_home_landed.bmp
assert_pixel 808 115 000000 60                # AC on foundation 2 (its middle pip); 2H followed AH
close

# ---- Enhanced animations: cards turning over, the dragged cards' shadow, the deal out of the stock ------

setenv SOLHD_ANIM_SLOW 10

sync                                          # deal 64
move_window 0 120
resize_client 1264 854
assert_client_size 1264 854
async on
click 100 100                                 # a draw; AH, AC, 2H home; cols 6 and 7 turn their next card
wait 900
capture 83_flip_mid_turn.bmp                  # (120 ms a turn, 1.2 s at 10 x)
async off
sync
capture 84_turned.bmp
assert_pixel 1000 300 FFFFFF                  # col 6's new top card: face up
ldown 277 400                                 # 4S (col 2)
move 400 500
move 640 600
capture 85_drag_shadow.bmp
async on
lup 640 600                                   # nowhere: it slides back, eased (with its shadow)
wait 250
capture 86_zip_back_mid_air.bmp
async off
sync
capture 87_zip_back_done.bmp
assert_pixel 277 443 000000                   # 4S back in col 2
async on
command 1000                                  # Deal: 28 cards out of the stock, 10 ms apart (100 at 10 x)
wait 1200
capture 88_deal_mid_air.bmp
async off
sync
capture 89_dealt.bmp
close
