# End-to-end smoke test of FreeCell HD (build/FreeCellHD.exe) under Wine.
#   make e2e        (or: tools/wine/run.sh -o build/e2e/fchd tests/e2e/fchd_smoke.txt)
# Ids: docs/xp-reference/resources.md + res/resource.h (103 Select Game, edit 203; 105 Statistics;
# 109 Options; 107 Restart; 115 Undo (F10); 116 Redo (Ctrl+Y, extra)).
# At a 632x427 client the scale is exactly 1, so every coordinate is XP's (layout.md section 2):
# columns x = 7 85 163 241 319 397 475 553 (71 wide), first card y = 106, step 18, card 71x96,
# free cells x = 0 71 142 213, home cells x = 348 419 490 561, top row y 0..95.
# Game #1 (rules.md section 1.3), columns top to bottom:
#   1 JD KD 2S 4C 3S 6D 6S   2 2D KC KS 5C TD 8S 9C   3 9H 9S 9D TS 4S 8D 2H   4 JC 5S QD QH TH QS 6H
#   5 5D AD JS 4H 8H 6C      6 7H QC AS AC 2C 3D      7 7C KH AH 4D JH 8C      8 5H 3H 3C 7S 7D TC

timeout 60000                             # start-up decode: seconds under Rosetta (x87 emulation)
regdel HKCU\Software\Microsoft\Windows\CurrentVersion\Applets\FreeCell
regdel HKCU\Software\xp-cards
launch ${REPO}/build/FreeCellHD.exe
info
assert_title FreeCell
assert_menu 107 grayed                    # Restart, Undo and Redo start grayed
assert_menu 115 grayed
assert_menu 116 grayed
assert_menubar_text "Cards Left: 0"
assert_reg HKCU\Software\Microsoft\Windows\CurrentVersion\Applets\FreeCell AlreadyPlayed 1   # migration ran
capture 00_startup.bmp
capture_window 00_startup_window.bmp

# ---- Select Game #1 through the dialog ----------------------------------------------------------------
command 103
wait_dialog "Game Number"
dialog_text
dialog_set_text 203 1
dialog_click OK
assert_no_dialog
assert_title "FreeCell Game #1"
assert_menubar_text "Cards Left: 52"
assert_menu 107 enabled
assert_menu 115 grayed
assert_menu 116 grayed

# ---- XP geometry at 632 x 427 -----------------------------------------------------------------------------
resize_client 632 427
assert_client_size 632 427
assert_pixel 0 0 000000                   # free cell 0 bevel: black top-left
assert_pixel 70 50 00FF00                 # ... green right edge
assert_pixel 297 30 00FF00                # king frame: green left
assert_pixel 334 30 000000                # ... black right
assert_pixel 5 300 007F00                 # table green left of column 1
capture 01_game1_632x427.bmp
capture_window 01_game1_632x427_window.bmp

# ---- selection = inverted card ---------------------------------------------------------------------------
assert_pixel 11 264 FFFFFF                # white margin of 6S, bottom card of column 1 (y 214..309)
click 42 262
assert_pixel 11 264 000000
capture 02_selected.bmp
click 42 262                              # click it again: deselect
assert_pixel 11 264 FFFFFF
assert_menu 115 grayed                    # a pure deselect is not an action

# illegal move with messages on: 6S onto 9C
click 42 262
click 120 300
wait_dialog FreeCell
assert_dialog_text "That move is not allowed."
dialog_click OK
assert_no_dialog
assert_pixel 11 264 FFFFFF                # deselected after the message

# ---- moves: 3D to free cell 0, then 2C to free cell 1 (autoplay sends AC, 2C, AS home) --------------

assert_pixel 4 50 FFFFFF                  # a card in free cell 0
assert_pixel 401 285 007F00               # gone from column 6 (2C now ends at y 273)
assert_menu 115 enabled
assert_menubar_text "Cards Left: 52"
capture 03_3D_to_freecell.bmp
click 430 240
click 106 48
assert_menubar_text "Cards Left: 49"
assert_title "FreeCell Game #1"
capture 04_autoplay.bmp

vkey Y ctrl

assert_menubar_text "Cards Left: 49"
assert_menu 116 grayed
assert_menu 115 enabled
capture 06_redo.bmp

# ---- an empty column and the Move to Empty Column dialog ------------------------------------------------

click 42 262                              # 6S (column 1) -> onto 7H
click 510 300
